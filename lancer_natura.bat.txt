@echo off
title Lancement de NATURA AI
cd /d "%~dp0"
echo Recherche de Python...

py --version >nul 2>&1
if %errorlevel% neq 0 (
    python --version >nul 2>&1
    if %errorlevel% neq 0 (
        echo.
        echo ERREUR : Python n'est pas reconnu par Windows.
        echo Veuillez installer la vraie version de Python sur python.org
        pause
        exit
    ) else (
        set PY_CMD=python
    )
) else (
    set PY_CMD=py
)

echo Creation de l'environnement virtuel...
%PY_CMD% -m venv .venv

echo Activation de l'environnement...
call .venv\Scripts\activate.bat

echo Installation des composants (cela peut prendre 2 minutes)...
pip install -r requirements.txt

echo Lancement de NATURA AI...
python -m streamlit run app.py

pause
